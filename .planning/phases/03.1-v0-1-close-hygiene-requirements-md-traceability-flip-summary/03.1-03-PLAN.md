   def test_health_never_rate_limited():
       """
       Phase 3.1 R-01: /health must NOT consume /search quota.

       The D-05 refactor moves rate limits to Limiter(default_limits=...) which
       applies to EVERY route by default. We exempt /health and /health/deep
       via @limiter.exempt so liveness probes don't burn the /search budget.
       This test pins that invariant — 70 GETs to /health (well above the
       60/minute /search budget) must all return 200, never 429.

       If this test fails, the @limiter.exempt decorator was lost during a
       refactor and the load balancer will think the service is down once
       it hits the /search quota.
       """
       with TestClient(app) as client:
           # Reset limiter to isolate from sibling-test state (mirrors the
           # pattern in tests/integration/test_rate_limit.py).

               pass

                   f"R-01 violated: /health 429'd on request {i} — "
                   f"@limiter.exempt is missing or default_limits applied. "
                   f"Body: {r.text[:200]}"

   def test_default_limits_propagation_matches_settings():
       """
       Phase 3.1 D-05 / R-A3: assert the runtime Limiter.default_limits match
       settings.API_RATE_PER_MINUTE/DAY exactly. This is the import-time
       invariant — if a future refactor disconnects the constants from
       settings, this test fires immediately.

       from src.artiscrapper.config import settings

       limiter = app.state.limiter
       defaults = list(limiter._default_limits)  # slowapi 0.1.9 internal attr

           f"D-05: expected 2 default_limits (per_minute + per_day); got {len(defaults)}: {defaults}"

       # Render each LimitGroup's `.limit` back to its string form, or
       # fall back to str() — both forms accepted by slowapi 0.1.9.
