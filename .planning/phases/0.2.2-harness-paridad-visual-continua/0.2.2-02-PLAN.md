<objective>
Wire the 1/10 sample audit into the production `/search` hot path. The sample
reuses the HTML the pipeline already produced — NO extra Google fetch — and
runs the parity-metric helper in a background task that never blocks the
response or raises into request handling.

**Purpose:** Without continuous sampling, the parity gauges only update when
someone manually hits `/admin/parity/{q}` or the nightly CI runs. The sample
gives Prometheus a continuous low-cost signal under real production load,
which is what the WARN/FAIL alerts in Plan 03 rely on.

**Output:**
- New setting `PARITY_SAMPLE_RATE` (default 0.1, env-tunable)
- `_parity_audit_sample(query, html)` async fn that runs the helper safely
- Module-level `_PARITY_BG_TASKS` strong-ref set
- POST /search schedules the sample with `random.random() < settings.PARITY_SAMPLE_RATE`
- Integration tests measuring overhead and verifying safety
</objective>

```python
import random
from selectolax.parser import HTMLParser as _ParityHTMLParser
from .search import parse_serp as _parity_parse_serp

# Phase 0.2.2 — fire-and-forget background tasks for parity sampling.
# WR-02 pattern: strong-ref set + add_done_callback prevents GC from
# collecting the task before completion (the bg cache-write uses the same
# pattern; see commit 2026-06-04).
_PARITY_BG_TASKS: set[asyncio.Task] = set()


async def _parity_audit_sample(query: str, html: str) -> None:
    """Run parity metrics on already-cached HTML (no Cloak fetch).
    Fire-and-forget — NEVER raises into the caller. Overhead expectation:
    <50ms p99 on a 1.5 MB SERP HTML (DOM parse + count regexes).

    Why we reuse `html` instead of fetching fresh:
    - `/search` already paid the cost of fetching it.
    - The parity metric of interest is "what's in THIS html that the parser
      missed", not "did Google's auction return more pla-units 200ms later".
    - Avoids doubling Cloak rate-limit consumption per /search call.
    """
    try:
        tree = _ParityHTMLParser(html)
        pla_nodes_count = len(tree.css("div.pla-unit"))
        candidates = _parity_parse_serp(html)
        _compute_parity_metrics(query, html, candidates, pla_nodes_count)
    except Exception as exc:
        log.warning("parity_sample_failed", query=query[:80], error=str(exc))
```

```python
"""Phase 0.2.2 HARNESS-03 — sample hot-path overhead + safety tests."""
import asyncio
import time
from pathlib import Path

import pytest

from src.artiscrapper.main import _parity_audit_sample, _PARITY_BG_TASKS

FIXTURE = (
    Path(__file__).parent.parent / "fixtures" / "serp" / "pla_unit_robotech.html"
)


@pytest.mark.asyncio
async def test_sample_p99_under_50ms_on_fixture():
    """Run _parity_audit_sample 30 times on the robotech fixture; p99 must be <50ms.

    Why 50ms: this is the budget for not impacting /search p99 noticeably even
    when sampling at 100% (worst case). At PARITY_SAMPLE_RATE=0.1 the real
    impact on average /search latency is /10.
    """
    html = FIXTURE.read_text()
    timings = []
    for _ in range(30):
        t0 = time.perf_counter()
        await _parity_audit_sample("benchmark query", html)
        timings.append((time.perf_counter() - t0) * 1000)  # ms
    timings.sort()
    p99 = timings[int(len(timings) * 0.99) - 1] if len(timings) >= 100 else timings[-1]
    median = timings[len(timings) // 2]
    print(f"\nparity sample on 726KB fixture: median={median:.1f}ms p99={p99:.1f}ms")
    assert p99 < 50.0, f"sample overhead too high: p99={p99:.1f}ms (>=50ms)"


@pytest.mark.asyncio
async def test_sample_never_raises_on_broken_html(caplog):
    """Even truly broken inputs must NOT raise — failure is logged silently."""
    # Empty / garbage / None-ish
    for bad in ["", "<<<not html>>>", "{}", " "]:
        await _parity_audit_sample("safety query", bad)
    # If we got here without raising, success.
    assert True


@pytest.mark.asyncio
async def test_sample_updates_metrics_on_real_fixture():
    """A successful sample call updates the parity gauges/counter."""
    from src.artiscrapper.metrics import (
        parity_pla_in_html,
        parity_pla_extracted,
    )
    html = FIXTURE.read_text()
    label = "metric update query"
    await _parity_audit_sample(label, html)
    pla_in_html = parity_pla_in_html.labels(query=label)._value.get()
    extracted = parity_pla_extracted.labels(query=label)._value.get()
    assert pla_in_html >= 4, f"Expected >=4 (frozen fixture), got {pla_in_html}"
    assert extracted >= 4
```
